# Mag7 Agent Implementation Plan

> **For agentic workers:** REQUIRED: Use superpowers:subagent-driven-development (if subagents available) or superpowers:executing-plans to implement this plan. Steps use checkbox (`- [ ]`) syntax for tracking.

**Goal:** Create a standalone OpenClaw `mag7` subagent that scans Magnificent 7 stocks every 15 minutes during market hours for volume spikes + technical confluence, sends WhatsApp approval requests, and executes paper trades.

**Architecture:** A Python scanner (`mag7_scanner.py`) runs via host cron every 15 min (Mon–Fri, 13:00–21:00 UTC) inside the moltbot Docker container, fetches 15-min OHLCV from Finnhub, detects volume spikes (current vol > 2× 20-bar avg), scores signals 0–5 via EMA/RSI/MACD/VWAP/BB, and writes a pending signal file if score ≥ 4. The `mag7` OpenClaw agent wakes on its 2-min HEARTBEAT, sees the pending signal, sends a WhatsApp approval message, and on "yes" opens a paper position tracked in `mag7-trading-state.json`. Exits at +5% gain or −2% stop-loss.

**Tech Stack:** Python 3, finnhub-python, numpy, pytz, pytest; OpenClaw agent docs (SOUL/BOOT/HEARTBEAT/IDENTITY.md); JSON state files; moltbot Docker container; WhatsApp channel (existing); macOS host cron.

    \"\"\"Return True if NYSE is currently open (Mon-Fri, 9:30am-4pm ET).

    return open_time <= now_et < close_time

        key = cfg.get(\"env\", {}).get(\"vars\", {}).get(\"FINNHUB_API_KEY\", \"\")

# ─── Volume Spike Detection ───────────────────────────────────────────────────

def detect_volume_spike(
    volumes: list,
    multiplier: float = SPIKE_MULTIPLIER,
    avg_bars: int = SPIKE_AVG_BARS,
    return_ratio: bool = False,
):
    """
    Returns True if current (last) volume > multiplier * avg of previous avg_bars.
    If return_ratio=True, returns the actual ratio (float) instead of bool.
    Returns False / 0.0 if insufficient data.
    """
    if len(volumes) < avg_bars + 2:
        return 0.0 if return_ratio else False
    current_vol = volumes[-1]
    avg_vol = sum(volumes[-avg_bars - 1:-1]) / avg_bars
    if avg_vol == 0:
        return 0.0 if return_ratio else False
    ratio = current_vol / avg_vol
    if return_ratio:
        return ratio
    return ratio > multiplier


# ─── Signal Scoring (0-5 points) ─────────────────────────────────────────────

def score_signal(
    closes: "np.ndarray",
    highs:  "np.ndarray",
    lows:   "np.ndarray",
    volumes:"np.ndarray",
    timestamps:"np.ndarray",
    min_bars: int = 30,
) -> int:
    """
    Score the current bar 0-5 using technical indicators.
    Requires at least min_bars of data.

    Scoring rules (each = +1):
      1. EMA9 > EMA21 (bullish EMA cross)
      2. RSI(14) between 40 and 70 (momentum, not overbought)
      3. MACD histogram > 0 (bullish momentum)
      4. Close > VWAP (price above intraday mean)
      5. Bollinger %B between 0.4 and 0.8 (mid-to-upper band, not extended)
    """
    if len(closes) < min_bars:
        return 0

    try:
        from indicators import ema, rsi, macd, bollinger_bands, vwap, OHLCV

        ohlcv = OHLCV(
            timestamp=timestamps,
            open=closes,     # approx open as close for scoring only
            high=highs,
            low=lows,
            close=closes,
            volume=volumes,
        )

        ema9_arr  = ema(closes, 9)
        ema21_arr = ema(closes, 21)
        rsi14_arr = rsi(closes, 14)
        macd_res  = macd(closes, 12, 26, 9)
        bb_res    = bollinger_bands(closes, 20, 2.0)
        vwap_arr  = vwap(ohlcv)

        score = 0
        idx = -1   # latest bar

        # 1. EMA cross
        if not (np.isnan(ema9_arr[idx]) or np.isnan(ema21_arr[idx])):
            if ema9_arr[idx] > ema21_arr[idx]:
                score += 1

        # 2. RSI in momentum zone
        if not np.isnan(rsi14_arr[idx]):
            if 40 <= rsi14_arr[idx] <= 70:
                score += 1

        # 3. MACD histogram positive
        if not np.isnan(macd_res.histogram[idx]):
            if macd_res.histogram[idx] > 0:
                score += 1

        # 4. Close above VWAP
        if not np.isnan(vwap_arr[idx]):
            if closes[idx] > vwap_arr[idx]:
                score += 1

        # 5. BB %B in mid-to-upper range (not exhausted)
        if not np.isnan(bb_res.percent_b[idx]):
            if 0.4 <= bb_res.percent_b[idx] <= 0.8:
                score += 1

        return score

    except Exception as e:
        import sys
        print(f"  [score_signal] Error: {e}", file=sys.stderr)
        return 0


'''

finnhub_and_scan = '''

# ─── Finnhub OHLCV Fetch ──────────────────────────────────────────────────────

def fetch_ohlcv_finnhub(ticker: str, api_key: str, bars: int = MIN_BARS + 5) -> Optional[dict]:
    """
    Fetch 15-min OHLCV candles from Finnhub.
    Returns dict with keys: t, o, h, l, c, v (lists) or None on failure.
    Retries 2x with 5s backoff on transient errors.
    """
    import finnhub
    try:
        client = finnhub.Client(api_key=api_key)
    except Exception as e:
        log(f"  [{ticker}] Finnhub client init failed: {e}")
        return None

    now_ts  = int(time.time())
    from_ts = now_ts - (bars * 15 * 60) - 3600  # extra buffer

    for attempt in range(3):
        try:
            res = client.stock_candles(ticker, FINNHUB_RESOLUTION, from_ts, now_ts)
            if res.get("s") == "ok" and res.get("c"):
                return res
            if res.get("s") == "no_data":
                log(f"  [{ticker}] No data from Finnhub")
                return None
            log(f"  [{ticker}] Finnhub status: {res.get('s', 'unknown')} — attempt {attempt+1}")
        except Exception as e:
            log(f"  [{ticker}] Finnhub error (attempt {attempt+1}): {e}")
        if attempt < 2:
            time.sleep(5)

    return None


def candles_to_arrays(candle_data: dict) -> tuple:
    """Convert Finnhub candle dict to numpy arrays: (timestamps_ms, opens, highs, lows, closes, volumes)."""
    t = np.array(candle_data["t"], dtype=float) * 1000   # s -> ms
    o = np.array(candle_data["o"], dtype=float)
    h = np.array(candle_data["h"], dtype=float)
    l = np.array(candle_data["l"], dtype=float)
    c = np.array(candle_data["c"], dtype=float)
    v = np.array(candle_data["v"], dtype=float)
    return t, o, h, l, c, v


# ─── Per-Ticker Scan ──────────────────────────────────────────────────────────

def scan_ticker(ticker: str, api_key: str) -> Optional[dict]:
    """
    Scan one ticker. Returns signal dict if spike+score threshold met, else None.
    """
    log(f"  Scanning {ticker}...")
    candles = fetch_ohlcv_finnhub(ticker, api_key)
    if candles is None:
        return None

    ts, opens, highs, lows, closes, volumes = candles_to_arrays(candles)

    if len(closes) < MIN_BARS:
        log(f"  [{ticker}] Only {len(closes)} bars — need {MIN_BARS}, skipping")
        return None

    vol_list  = list(volumes)
    spike     = detect_volume_spike(vol_list)
    vol_ratio = detect_volume_spike(vol_list, return_ratio=True)

    if not spike:
        log(f"  [{ticker}] No spike (ratio={vol_ratio:.2f}x)")
        return None

    score = score_signal(closes, highs, lows, volumes, ts)
    current_price = float(closes[-1])

## Key Design Decisions

| Decision | Reasoning |
|----------|-----------|
| Cron 13-21 UTC (not 13-20) | Covers both EDT (UTC-4) and EST (UTC-5) NYSE close; Python gate handles exact 9:30am open |
| `notified_at` for expiry, not `ts` | User has 10 min from when they *received* the WhatsApp, not when scanner fired |
| Skipped signals → `skippedSignals` list | Keeps `closedTrades` clean for P&L calculations |
| `type: token` auth format | Matches what refresh-anthropic-token.sh injects; OpenClaw uses as plain bearer |
| 2× 20-bar avg for spike | Conservative; filters noise while catching real institutional volume on 15-min candles |
| Score ≥ 4/5 to enter | High conviction; adapts btc-trader's selectivity philosophy |
| Atomic rename for signal file | Prevents partial reads if scanner crashes mid-write |
| Scanner reads key from openclaw.json | Centralizes config; no separate env file needed for cron |
| git init for workspace | Required by OpenClaw for session compaction (matches btc-trader pattern) |
